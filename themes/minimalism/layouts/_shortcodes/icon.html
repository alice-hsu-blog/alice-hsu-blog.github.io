{{- /* 舊主題的圖示 shortcode。assets/icons/ 有同名 SVG 時畫圖示，沒有時退回文字；有 url 時包成連結。 */ -}} {{- $name := .Get "name" | default (.Get 0) -}} {{- $url := .Get "url" | default (.Get "href") -}} {{- $label := .Get "label" | default (.Get "ariaLabel") | default (.Get "title") | default $name -}} {{- $newtab := in (slice true "true" 1 "1") (.Get "newtab") -}} {{- $svg := resources.Get (printf "icons/%s.svg" $name) -}} {{- if $url -}} {{ if $svg }}{{ $svg.Content | safeHTML }}{{ else }}{{ $label }}{{ end }} {{- else if $svg -}} {{ $svg.Content | safeHTML }} {{- else -}} {{ $label }} {{- end -}}
